## 1. Overview

### Current CI State

The repository currently has two GitHub Actions workflows:

| Workflow | File | Purpose |
|----------|------|---------|
| E2E Tests | `.github/workflows/e2e.yml` | Builds Leptos app, starts Node.js mock backend (`npm ci` + `node server.js`), runs Playwright E2E tests |
| Security Scan | `.github/workflows/security.yml` | Gitleaks secrets scan, Trivy vulnerability scan, auto-labelling, branch sync checks, Discord notifications |

**Key Node.js dependencies in current E2E pipeline:**
- `actions/setup-node@v4` (Node.js 20)
- `npm ci` in `packages/mock_backend/`
- `node server.js` spawned by Playwright `global-setup.ts`
- `tree-kill` for process teardown in `global-teardown.ts`

### Target State

After this plan:

1. **New standalone CI job** — `cargo test`, `cargo clippy`, `cargo fmt --check` for `packages/mock_backend`
2. **Updated E2E workflow** — Replace Node.js mock backend with Rust binary (`cargo run`) or in-process tower-based testing
3. **SDL snapshot testing** — Catch accidental schema drift via `insta` snapshots
4. **Eliminated Node.js dependency** — No `npm ci`, `setup-node`, or Node.js runtime needed for the mock backend
5. **Cargo workspace integration** — Optional: add `packages/mock_backend` to a Cargo workspace for unified builds

```yaml
on:
  pull_request:
    paths:
      - "packages/mock_backend/**"

  push:
    branches: [main]
    paths:
      - "packages/mock_backend/**"

    name: Check, Lint & Test

    timeout-minutes: 15

      run:
        working-directory: packages/mock_backend

    steps:
      - uses: actions/checkout@v4

      - name: Install Rust toolchain
        uses: dtolnay/rust-toolchain@stable
        with:
          components: rustfmt, clippy

      - name: Cache Rust build
        uses: Swatinem/rust-cache@v2
        with:
          workspaces: packages/mock_backend -> target

      - name: Check formatting
        run: cargo fmt --check

      - name: Run Clippy
        run: cargo clippy --all-targets -- -D warnings

      - name: Run tests
        run: cargo test --all-targets

      - name: Build release
        run: cargo build --release

  snapshot:
    name: Schema Snapshot
    runs-on: ubuntu-latest
    timeout-minutes: 10
    needs: check

    defaults:
      run:
        working-directory: packages/mock_backend

    steps:
      - uses: actions/checkout@v4

      - name: Install Rust toolchain
        uses: dtolnay/rust-toolchain@stable

      - name: Cache Rust build
        uses: Swatinem/rust-cache@v2
        with:
          workspaces: packages/mock_backend -> target

      - name: Install cargo-insta
        run: cargo install cargo-insta

      - name: Verify schema snapshots
        run: cargo insta test --review=fail
```

```rust
/// Assert that the mock backend's GraphQL SDL has not changed unexpectedly.
```

### 4.5 Updated `global-setup.ts`

Replace the Node.js mock backend spawn with the compiled Rust binary:

```typescript
import { FullConfig } from "@playwright/test";
import { spawn, ChildProcess } from "child_process";
import path from "path";
import fs from "fs";
import { waitForServer } from "./helpers/wait-for-server";

const PID_FILE = path.join(__dirname, ".test-pids.json");

interface ProcessPids {
  mockBackend: number;
  leptosApp: number;
}

/**
 * Resolve the mock backend binary path.
 *
 * Priority:
 * 1. MOCK_BACKEND_BIN env var (set by CI)
 * 2. Pre-compiled release binary at standard cargo output path
 * 3. Fallback: spawn via `cargo run --release` (slower, compiles on the fly)
 */
function resolveMockBackend(): { command: string; args: string[]; cwd: string } {
  const envBin = process.env.MOCK_BACKEND_BIN;
  if (envBin) {
    const resolved = path.resolve(__dirname, envBin);
    if (fs.existsSync(resolved)) {
      return { command: resolved, args: [], cwd: path.dirname(resolved) };
    }
  }

  const releaseBin = path.resolve(
    __dirname,
    "../../packages/mock_backend/target/release/mock_backend"
  );
  if (fs.existsSync(releaseBin)) {
    return { command: releaseBin, args: [], cwd: path.dirname(releaseBin) };
  }

  // Fallback: cargo run (will compile if needed)
  return {
    command: "cargo",
    args: ["run", "--release"],
    cwd: path.resolve(__dirname, "../../packages/mock_backend"),
  };
}

async function globalSetup(config: FullConfig) {
  if (process.env.SKIP_GLOBAL_SETUP) {
    console.log("⏭️  Skipping global setup (SKIP_GLOBAL_SETUP=true)");
    return;
  }

  console.log("\n🔧 Starting mock backend (Rust)...");

  const mock = resolveMockBackend();
  const mockBackend = spawn(mock.command, mock.args, {
    cwd: mock.cwd,
    stdio: "pipe",
    detached: true,
    env: { ...process.env, PORT: "4000" },
  });

  mockBackend.stdout?.on("data", (data: Buffer) => {
    if (process.env.DEBUG) console.log(`[mock] ${data.toString().trim()}`);
  });
  mockBackend.stderr?.on("data", (data: Buffer) => {
    console.error(`[mock:err] ${data.toString().trim()}`);
  });

  await waitForServer(4000, 30_000);
  console.log("✅ Mock backend ready on :4000");

  console.log("🔧 Starting Leptos app...");

  const leptosApp = spawn("cargo", ["leptos", "serve", "--release"], {
    cwd: path.resolve(__dirname, "../"),
    stdio: "pipe",
    detached: true,
    env: {
      ...process.env,
      GRAPHQL_ENDPOINT: "http://localhost:4000/graphql",
    },
  });

  leptosApp.stdout?.on("data", (data: Buffer) => {
    if (process.env.DEBUG) console.log(`[leptos] ${data.toString().trim()}`);
  });
  leptosApp.stderr?.on("data", (data: Buffer) => {
    if (process.env.DEBUG) console.log(`[leptos] ${data.toString().trim()}`);
  });

  await waitForServer(3000, 120_000);
  console.log("✅ Leptos app ready on :3000");

  // Save PIDs for teardown
  const pids: ProcessPids = {
    mockBackend: mockBackend.pid!,
    leptosApp: leptosApp.pid!,
  };
  fs.writeFileSync(PID_FILE, JSON.stringify(pids));
  mockBackend.unref();
  leptosApp.unref();
}

export default globalSetup;
```

If a root Cargo workspace is preferred:

**Trade-offs:**

| Aspect | Workspace | Independent crates |
|--------|-----------|-------------------|
| Build caching | Shared `target/` dir — faster incremental builds | Separate `target/` dirs — no cross-contamination |
| CI commands | `cargo test --workspace` runs everything | Separate `cd` + `cargo test` per crate |
| Dependency unification | Cargo resolves a single version per dep | Each crate can pin independently |
| IDE support | rust-analyzer loads everything at once (heavier) | Lighter per-crate analysis |

**Recommendation:** Keep crates independent for now. The mock backend is a test fixture with different lifecycle from the Leptos app. A workspace can be added later if build times benefit from shared compilation.

|---|------|--------|

Example implementation:

```rust
// tests/mock_integration.rs

use mock_backend::{app, default_shared_state, app_with_state};
use axum::body::Body;
use http::Request;
use tower::ServiceExt;
use serde_json::{json, Value};

async fn graphql_request(app: axum::Router, query: &str) -> Value {
    let body = json!({ "query": query });
    let request = Request::builder()
        .method("POST")
        .uri("/graphql")
        .header("Content-Type", "application/json")
        .body(Body::from(serde_json::to_string(&body).unwrap()))
        .unwrap();

    let response = app.oneshot(request).await.unwrap();
    let body_bytes = axum::body::to_bytes(response.into_body(), usize::MAX)
        .await
        .unwrap();
    serde_json::from_slice(&body_bytes).unwrap()
}

#[tokio::test]
async fn test_mock_health_check() {
    let app = app();
    let res = graphql_request(app, "{ _health }").await;
    assert_eq!(res["data"]["_health"], true);
}

#[tokio::test]
async fn test_mock_reset() {
    let app = app();
    let request = Request::builder()
        .method("POST")
        .uri("/reset")
        .body(Body::empty())
        .unwrap();
    let response = app.oneshot(request).await.unwrap();
    assert_eq!(response.status(), 200);
}
```
